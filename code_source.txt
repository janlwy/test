    private function addMusic($post, $files)

                    "Extension de fichier audio non autorisÃ©e.\nExtension : %s\nExtensions autorisÃ©es : %s",

                );

                    "Type MIME audio non autorisÃ©.\nType dÃ©tectÃ© : %s\nTypes autorisÃ©s : %s",
                    $audioMimeType,

                );

                break;
        }

            $this->addError($field, $message ?? $defaultMessage);
        }

    private function validateParameterizedRule($field, $value, $ruleName, $parameter, $message = null) {
        // Si la valeur est vide et que ce n'est pas une rÃ¨gle 'required', on ne valide pas
        if (($value === null || $value === '') && $ruleName !== 'required') {
            return;
        }
        
        $isValid = true;
        $defaultMessage = '';
        
        switch ($ruleName) {
            case 'min':
                $isValid = strlen($value) >= $parameter;
                $defaultMessage = "Minimum $parameter caractÃ¨res requis";
                break;
                
            case 'max':
                $isValid = strlen($value) <= $parameter;
                $defaultMessage = "Maximum $parameter caractÃ¨res autorisÃ©s";
                break;
                
            case 'match':
                $matchValue = $this->data[$parameter] ?? null;
                $isValid = $value === $matchValue;
                $defaultMessage = "Les valeurs ne correspondent pas";
                break;
                
            case 'pattern':
                $isValid = preg_match($parameter, $value);
                $defaultMessage = "Format invalide";
                break;
                
            case 'mime':
                $isValid = in_array($value, (array)$parameter);
                $defaultMessage = "Type de fichier non autorisÃ©";
                break;
                
            case 'maxsize':
                $isValid = $value <= $parameter;
                $defaultMessage = "Taille maximale dÃ©passÃ©e";
                break;
        }
        
        if (!$isValid) {
            $this->addError($field, $message ?? $defaultMessage);
        }
    }

        public function updateTable($table, $data, $id) {
            if (empty($table) || empty($data) || !is_array($data) || !is_numeric($id)) {
                throw new DatabaseException("ParamÃ¨tres invalides pour la mise Ã  jour");
            }

            try {
                $this->beginTransaction();
                $connexion = $this->getConnexion();
                
                // VÃ©rifier si l'enregistrement existe
                $checkSql = "SELECT id FROM " . $this->escapeIdentifier($table) . " WHERE id = :id";
                $checkStmt = $connexion->prepare($checkSql);
                $checkStmt->bindValue(':id', $id, PDO::PARAM_INT);
                $checkStmt->execute();
                
                if ($checkStmt->rowCount() === 0) {
                    throw new DatabaseException("L'enregistrement avec l'ID $id n'existe pas dans la table $table");
                }
                
                $updates = [];
                $params = [];
                
                foreach ($data as $column => $value) {
                    if ($column !== 'id') { // Ignorer la colonne id dans les donnÃ©es Ã  mettre Ã  jour
                        $updates[] = $this->escapeIdentifier($column) . " = :$column";
                        $params[$column] = $value;
                    }
                }
                
                if (empty($updates)) {
                    throw new DatabaseException("Aucune donnÃ©e valide Ã  mettre Ã  jour");
                }
                
                $sql = "UPDATE " . $this->escapeIdentifier($table) . 
                       " SET " . implode(", ", $updates) . 
                       " WHERE id = :id";
                
                $statement = $connexion->prepare($sql);
                $statement->bindValue(":id", $id, PDO::PARAM_INT);
                
                foreach ($params as $key => $value) {
                    $statement->bindValue(":$key", $value, $this->getParamType($value));
                }
                
                if (!$statement->execute()) {
                    throw new DatabaseException("Erreur lors de la mise Ã  jour");
                }
                
                $this->commit();
                return $this->readTableOne($table, $id);
                
            } catch (PDOException $e) {
                $this->rollback();
                logError("Erreur lors de la mise Ã  jour de la table $table: " . $e->getMessage());
                throw new DatabaseException("Erreur lors de la mise Ã  jour des donnÃ©es", 0, $e);
            } catch (\Exception $e) {
                $this->rollback();
                logError("Exception gÃ©nÃ©rique lors de la mise Ã  jour de la table $table: " . $e->getMessage());
                throw new DatabaseException("Erreur lors de la mise Ã  jour des donnÃ©es", 0, $e);
            }
        }

namespace Models;

class UserRole {
    public const ROLE_USER = 'user';
    public const ROLE_ADMIN = 'admin';
    
    private static $roleHierarchy = [
        self::ROLE_ADMIN => [self::ROLE_USER],
        self::ROLE_USER => []
    ];
    
    public static function hasRole(string $userRole, string $requiredRole): bool {
        if ($userRole === $requiredRole) {
            return true;
        }
        
        return in_array($requiredRole, self::$roleHierarchy[$userRole] ?? []);
    }
    
    public static function getAllowedRoutes(string $role): array {
        $routes = [
            self::ROLE_USER => [
                'mediabox', 'audio', 'video', 'photo', 'texte', 
                'compte', 'connexion', 'deconnexion'
            ],
            self::ROLE_ADMIN => [
                'admin', 'database_manager'
            ]
        ];
        
        $allowedRoutes = $routes[$role] ?? [];
        if (isset(self::$roleHierarchy[$role])) {
            foreach (self::$roleHierarchy[$role] as $inheritedRole) {
                $allowedRoutes = array_merge($allowedRoutes, $routes[$inheritedRole] ?? []);
            }
        }
        
        return array_unique($allowedRoutes);
    }
}
